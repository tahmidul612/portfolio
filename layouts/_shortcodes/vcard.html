{{- /* Renders the downloadable contact card. Identity is parsed out of the .vcf itself so the page and the file it serves can never drift apart. Update vcard.vcf and this follows. Params: src path to the vCard, relative to the project root href URL the card links to download filename browsers save it as */ -}} {{- $src := .Get "src" | default "static/vcard.vcf" -}} {{- $href := .Get "href" | default "/vcard" -}} {{- $download := .Get "download" | default "Tahmidul_Islam.vcf" -}} {{- $name := "" -}} {{- $role := "" -}} {{- $org := "" -}} {{- with (readFile $src) -}} {{- range (split . "\n") -}} {{- $line := chomp . -}} {{- if hasPrefix $line "FN:" -}} {{- $name = strings.TrimPrefix "FN:" $line -}} {{- else if hasPrefix $line "TITLE:" -}} {{- $role = strings.TrimPrefix "TITLE:" $line -}} {{- else if hasPrefix $line "ORG:" -}} {{- /* ORG may carry organisational units after a semicolon */ -}} {{- $org = index (split (strings.TrimPrefix "ORG:" $line) ";") 0 -}} {{- end -}} {{- end -}} {{- else -}} {{- errorf "[vcard] Cannot read %q. Check the src parameter." $src -}} {{- end -}} {{- /* vCard escapes commas and semicolons with a backslash */ -}} {{- $unescape := dict "\\," "," "\\;" ";" -}} {{- range $from, $to := $unescape -}} {{- $name = replace $name $from $to -}} {{- $role = replace $role $from $to -}} {{- $org = replace $org $from $to -}} {{- end -}} {{- /* not-prose opts out of Congo's link-highlight hover, which would otherwise flood the whole card with the primary colour */ -}} {{ $name }} {{- with $role }}{{ . }}{{ end }} {{- with $org }}{{ . }}{{ end }} Save contact card .vcf
